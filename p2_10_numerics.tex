\begin{frame}{Verifying derivatives: the Taylor test}
  For random $u$, $h$ and $t \to 0$:
  \begin{align*}
    \abs{f(u + th) - f(u) - t\,f'(u)h} &= O(t^2),\\
    \abs{f(u + th) - f(u) - t\,f'(u)h - \tfrac{t^2}{2} f''(u)[h,h]} &= O(t^3).
  \end{align*}
  \begin{itemize}
    \item Plot errors vs.\ $t$ on a log--log scale: slopes 2 and 3.
    \item Detects sign errors, wrong boundary/terminal conditions, inconsistent discrete adjoints.
  \end{itemize}
