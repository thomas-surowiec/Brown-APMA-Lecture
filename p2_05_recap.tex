\section{Recap: What We Can Compute}

\begin{frame}{Recap: the reduced problem and its oracle}
  \[
    \min_{u \in \Uad}\; f(u), \qquad \Uad = \{ u \in U : u_a \le u \le u_b \}.
  \]
  \begin{center}
  \renewcommand{\arraystretch}{1.3}
  \begin{tabular}{@{}lll@{}}
    \toprule
    Quantity & How & Cost (PDE solves) \\
    \midrule
    $f(u)$ & state solve & 1 (nonlinear) \\
    $\nabla f(u)$ & $+$ adjoint solve & $+1$ (linear) \\
    $\nabla^2 f(u)h$ & linearized $+$ second adjoint & $+2$ (linear) \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{1ex}
  \begin{block}{Optimality}
    $\ub = \Proj\big(\ub - s\nabla f(\ub)\big)$ for some (all) $s > 0$.
  \end{block}
  This fixed-point characterization drives everything in Part II.
\end{frame}

\begin{frame}{The Variational Inequality in Pointwise Form}
  \begin{hlbox}
    \textbf{Pointwise form.} For box constraints, the VI holds if and only if, for a.e.\ $x \in D$,
    \[
      \big(\bar p(x) + \alpha\bar z(x)\big)\big(v - \bar z(x)\big) \ge 0 \qquad \forall v \in [z_a(x), z_b(x)] .
    \]
  \end{hlbox}

  \visible<2->{
  \begin{hlbox}
    \textbf{Projection formula} ($\alpha > 0$):
    \(
      \bar z = P_{[z_a,z_b]}\big(-\tfrac{1}{\alpha}\bar p\big) = \max\big\{z_a, \min\{z_b, -\tfrac{1}{\alpha}\bar p\}\big\} \;\text{a.e.}
    \)
  \end{hlbox}
  }

  \visible<3->{
  \begin{hlbox}
    \textbf{KKT form.} With $\mu_a := (\bar p + \alpha\bar z)^+$ and $\mu_b := (\bar p + \alpha\bar z)^-$:
    \[
      \bar p + \alpha\bar z - \mu_a + \mu_b = 0, \qquad \mu_a,\mu_b \ge 0, \qquad
      \mu_a(\bar z - z_a) = \mu_b(z_b - \bar z) = 0 .
    \]
  \end{hlbox}
  }

  \visible<4->{
  \begin{qbox}
    With $s = 1/\alpha$: \; $\bar z = P_{Z_{\rm ad}}\big(\bar z - s\nabla\hat J(\bar z)\big)$, the starting point for projected gradients.
  \end{qbox}
  }
\end{frame}
