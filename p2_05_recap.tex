\section{Recap of Part I}

\begin{frame}{Recap of Part I}
  \begin{hlbox}
    \textbf{Reduced problem} (Sections 2--3): \; $\min_{z \in Z_{\rm ad}} \hat J(z) := \tfrac12 \norm{CSz - u_d}_{L^2}^2 + \tfrac{\alpha}{2}\norm{z}_{L^2}^2$. For $\alpha > 0$, $\hat J$ is $\alpha$-strongly convex with a unique minimizer $\bar z$.
  \end{hlbox}

  \visible<2->{
  \begin{hlbox}
    \textbf{Optimality system} (Section 4). $\bar z$ is optimal if and only if
    \[
      \begin{aligned}
        -\Delta \bar u &= \bar z \;\;\text{in } D, \quad \bar u = 0 \;\;\text{on } \partial D && \text{(state)}\\
        -\Delta \bar p &= \bar u - u_d \;\;\text{in } D, \quad \bar p = 0 \;\;\text{on } \partial D && \text{(adjoint)}\\
        (\bar p + \alpha \bar z,\; z - \bar z)_{L^2} &\ge 0 \quad \forall z \in Z_{\rm ad} && \text{(VI)}
      \end{aligned}
    \]
  \end{hlbox}
  }

  \visible<3->{
  \begin{qbox}
    \textbf{At any $z$} (Section 5): one state solve and one adjoint solve give $\nabla\hat J(z) = p + \alpha z$. How do we turn this into a mesh-independent algorithm?
  \end{qbox}
  }
\end{frame}

% ------------------------------------------------------------
% From the VI to pointwise conditions
% ------------------------------------------------------------
\begin{frame}{From the Variational Inequality to Pointwise Conditions}
  \begin{qbox}
    \textbf{Lemma.} With $g := \bar p + \alpha \bar z$, the VI $(g, z - \bar z)_{L^2} \ge 0$ $\forall z \in Z_{\rm ad}$ holds if and only if, a.e.\ in $D$,
    \[
      g \;\begin{cases}
        = 0 & \text{where } z_a < \bar z < z_b,\\
        \ge 0 & \text{where } z_a = \bar z < z_b,\\
        \le 0 & \text{where } z_a < \bar z = z_b.
      \end{cases}
    \]
  \end{qbox}

  \visible<2->{
  \begin{hlbox}
    \textbf{(i) Localize.} $Z_{\rm ad}$ is defined pointwise: for $w \in Z_{\rm ad}$ and measurable $M \subset D$, $\bar z + \chi_M (w - \bar z) \in Z_{\rm ad}$. So $\int_M g\,(w - \bar z) \dd x \ge 0$ for all $M$, i.e.\ $g\,(w - \bar z) \ge 0$ a.e.
  \end{hlbox}
  }

  \visible<3->{
  \begin{hlbox}
    \textbf{(ii) Test with the bounds} $z_a, z_b \in Z_{\rm ad}$: \; $g \ge 0$ where $\bar z < z_b$, \; $g \le 0$ where $\bar z > z_a$.\\[0.3ex]
    \visible<4->{\textbf{(iii) Converse.} The cases give $g\,(z - \bar z) \ge 0$ a.e.\ for all $z \in Z_{\rm ad}$; integrate. \hfill \qed}
  \end{hlbox}
  }
\end{frame}

% ------------------------------------------------------------
% The projection formula
% ------------------------------------------------------------
\begin{frame}{The Projection Formula}
  \begin{columns}[c]
    \begin{column}{0.56\textwidth}
      \begin{hlbox}
        For $\alpha > 0$, divide the three cases by $\alpha$: with $w := -\tfrac{1}{\alpha}\bar p$,
        \[
          \begin{aligned}
            \bar z &= w && \text{where } z_a < \bar z < z_b,\\
            w &\le z_a && \text{where } \bar z = z_a,\\
            w &\ge z_b && \text{where } \bar z = z_b .
          \end{aligned}
        \]
      \end{hlbox}
    \end{column}
    \begin{column}{0.40\textwidth}
      \begin{center}
      \begin{tikzpicture}[scale=0.8]
        \draw[-{Stealth[length=2mm]}, muted] (-0.3,0) -- (3.8,0) node[right, font=\footnotesize] {$w$};
        \draw[-{Stealth[length=2mm]}, muted] (0,-0.3) -- (0,3.0) node[above, font=\footnotesize] {$\bar z$};
        \draw[dashed, muted!60] (0,0.8) -- (3.6,0.8);
        \draw[dashed, muted!60] (0,2.2) -- (3.6,2.2);
        \node[left, font=\footnotesize] at (0,0.8) {$z_a$};
        \node[left, font=\footnotesize] at (0,2.2) {$z_b$};
        \draw[very thick, accent] (0.1,0.8) -- (0.8,0.8) -- (2.2,2.2) -- (3.5,2.2);
      \end{tikzpicture}
      \end{center}
    \end{column}
  \end{columns}

  \visible<2->{
  \begin{qbox}
    $\bar z = P_{[z_a,z_b]}\big(-\tfrac{1}{\alpha}\bar p\big) = \max\big\{z_a, \min\{z_b, -\tfrac{1}{\alpha}\bar p\}\big\}$ a.e., \; i.e.\ \; $\bar z = P_{Z_{\rm ad}}\big(\bar z - \tfrac{1}{\alpha}\nabla\hat J(\bar z)\big)$.
  \end{qbox}
  }
\end{frame}
