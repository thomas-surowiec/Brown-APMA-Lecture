\section{Appendix: The Abstract Setting}

\begin{frame}[label=abstract]{Abstract Setting: Standing Assumptions}
  \begin{hlbox}
    State space $U$ and residual space $W$ (Banach), control space $Z$ (Hilbert):
    \[
      \min_{(u,z) \in U \times Z}\; J(u,z) \quad\text{subject to}\quad e(u,z) = 0 \;\text{in } W, \quad z \in Z_{\rm ad}.
    \]
  \end{hlbox}

  \begin{hlbox}
    \begin{enumerate}
      \item[(A1)] $Z_{\rm ad} \subset Z$ is nonempty, convex, and closed.
      \item[(A2)] $J : U \times Z \to \R$ and $e : U \times Z \to W$ are continuously Fr\'echet differentiable.
      \item[(A3)] For each $z \in Z_{\rm ad}$ there is a unique $u = S(z) \in U$ with $e(S(z),z) = 0$.
      \item[(A4)] $e_u(S(z),z) \in \mathcal{L}(U,W)$ has a bounded inverse.
    \end{enumerate}
  \end{hlbox}

  \begin{qbox}
    \textbf{Model problem:} $U = \Hzero$, $W = H^{-1}(D)$, $Z = \Ltwo$, \; $e(u,z) = Au - \iota z$.
  \end{qbox}
  \vfill\hfill\hyperlink{general<3>}{\beamerreturnbutton{Back to Section 5}}
\end{frame}

\begin{frame}{Abstract Setting: Consequences of the Implicit Function Theorem}
  \begin{hlbox}
    Under (A2)--(A4), $S : Z \to U$ is $C^1$ with $S'(z)h = -e_u^{-1}e_z h$; it is $C^2$ if $J$ and $e$ are.
  \end{hlbox}

  \begin{hlbox}
    \begin{itemize}
      \item The reduced problem $\min_{z \in Z_{\rm ad}} \hat J(z)$, $\hat J(z) := J(S(z),z)$, is a smooth problem over a convex set.
      \item By Step 4, every local solution $\bar z$ satisfies $\hat J'(\bar z)(z - \bar z) \ge 0$ for all $z \in Z_{\rm ad}$.
      \item Without convexity of $\hat J$, this is only \emph{necessary}, and solutions need not be unique.
    \end{itemize}
  \end{hlbox}
  \vfill\hfill\hyperlink{general<3>}{\beamerreturnbutton{Back to Section 5}}
\end{frame}

\begin{frame}{Abstract Setting: Existence}
  \begin{qbox}
    \textbf{Theorem.} Assume (A1)--(A3) and
    \begin{itemize}
      \item $Z_{\rm ad}$ is bounded, or $\hat J$ is coercive on $Z_{\rm ad}$;
      \item $S$ is weak-to-strong continuous: $z_n \weakto z$ in $Z$ $\Rightarrow$ $S(z_n) \to S(z)$ in $U$;
      \item $J$ is continuous in $u$, and convex and continuous in $z$.
    \end{itemize}
    Then the reduced problem has at least one solution.
  \end{qbox}

  \begin{hlbox}
    \textbf{Proof.} Route B of Section 3, word for word: the tracking part converges by compactness of $S$, the control part is weakly l.s.c.\ by convexity. \hfill \qed
  \end{hlbox}

  \begin{hlbox}
    Weak-to-strong continuity typically comes from a priori bounds on $S(z)$ plus a compact embedding, exactly as in Section 2.
  \end{hlbox}
  \vfill\hfill\hyperlink{general<3>}{\beamerreturnbutton{Back to Section 5}}
\end{frame}

\begin{frame}{Abstract Setting: The Lagrangian Viewpoint}
  \begin{hlbox}
    With $\Lag(u,z,p) := J(u,z) - \dual{p}{e(u,z)}_{W^*,W}$, the optimality system reads
    \[
      \begin{aligned}
        \Lag_p(\bar u,\bar z,\bar p) &= 0 &&\Longleftrightarrow\quad e(\bar u,\bar z) = 0 && \text{(state)}\\
        \Lag_u(\bar u,\bar z,\bar p) &= 0 &&\Longleftrightarrow\quad e_u^*\bar p = J_u && \text{(adjoint)}\\
        \Lag_z(\bar u,\bar z,\bar p)(z - \bar z) &\ge 0 &&\Longleftrightarrow\quad \hat J'(\bar z)(z-\bar z) \ge 0 && \text{(VI)}
      \end{aligned}
    \]
  \end{hlbox}

  \begin{hlbox}
    \begin{itemize}
      \item $\hat J(z) = \Lag(S(z),z,p)$ for \emph{every} $p$, since $e(S(z),z) = 0$.
      \item Choosing $p$ with $\Lag_u = 0$ removes the term containing $S'(z)$: this is the adjoint equation.
    \end{itemize}
  \end{hlbox}
  \vfill\hfill\hyperlink{general<3>}{\beamerreturnbutton{Back to Section 5}}
\end{frame}

\begin{frame}{Abstract Setting: A Semilinear Example}
  \begin{hlbox}
    Example 3 with $\kappa \equiv 1$, $d \le 3$, $R \in C^1$ nondecreasing with polynomial growth, $0 \le z_a \le z \le z_b$:
    \(
      e(u,z) := -\Delta u + z\,R(u) - f .
    \)
  \end{hlbox}

  \begin{hlbox}
    \begin{itemize}
      \item (A3): $z \ge 0$ and $R$ nondecreasing make $u \mapsto -\Delta u + zR(u)$ monotone (Minty--Browder).
      \item (A4): $e_u\,\delta u = -\Delta\delta u + zR'(u)\,\delta u$ with $zR'(u) \ge 0$ is coercive, hence invertible.
      \item Adjoint: \; $-\Delta p + zR'(u)\,p = u - u_d$, \; $p|_{\partial D} = 0$.
      \item Gradient in $\Ltwo$: \; $\nabla \hat J(z) = \alpha z - R(u)\,p$, since $e_z h = h\,R(u)$.
    \end{itemize}
  \end{hlbox}

  \begin{qbox}
    $\hat J$ is \textbf{nonconvex}: the optimality system is only necessary.
  \end{qbox}
  \vfill\hfill\hyperlink{general<3>}{\beamerreturnbutton{Back to Section 5}}
\end{frame}
