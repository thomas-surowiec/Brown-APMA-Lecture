\section{Projected Newton Methods}

% ------------------------------------------------------------
\begin{frame}{Why Not Just Project the Newton Step?}
  \begin{columns}[c]
    \begin{column}{0.56\textwidth}
      \begin{hlbox}
        \textbf{Naive:} $z_{k+1} = P_{Z_{\rm ad}}\big(z_k - H^{-1}\nabla\hat J(z_k)\big)$.\\[0.5ex]
        \visible<2->{\textbf{Example.} $\hat J(z) = \tfrac12 (z - z_u)^\top Q (z - z_u)$, \; $Q = \begin{psmallmatrix} 1 & 0.9\\ 0.9 & 1\end{psmallmatrix}$, \; $z_u = (-1, 1)$, \; $Z_{\rm ad} = \{z_1 \ge 0\}$.
        Solution: $\bar z = (0, 0.1)$.}
      \end{hlbox}
    \end{column}
    \begin{column}{0.40\textwidth}
      \visible<2->{
      \begin{center}
      \begin{tikzpicture}[scale=1.1]
        \begin{scope}
          \clip (-1.9,-0.35) rectangle (1.0,1.8);
          \fill[brownred!7] (0,-0.35) rectangle (1.0,1.8);
          \draw[muted!50, rotate around={-45:(-1,1)}] (-1,1) ellipse (0.632 and 0.145);
          \draw[muted!80, rotate around={-45:(-1,1)}] (-1,1) ellipse (1.378 and 0.316);
          \draw[accent!70, rotate around={-45:(-1,1)}] (-1,1) ellipse (3.162 and 0.725);
        \end{scope}

        \fill[black] (0,0.1) circle (0.045);

        \visible<3->{

          \draw[-{Stealth[length=2mm]}, thick, accent] (-1,1) -- (-0.06,1);

      }
    \end{column}
  \end{columns}

  \visible<3->{
  \begin{qbox}
    Started \emph{at the solution}, Newton goes to $z_u$; projecting gives $\hat J = 0.5 > 0.095 = \hat J(\bar z)$: the coupling $Q_{12}$ between the active $z_1$ and the free $z_2$ drags $z_2$ away.
  \end{qbox}
  }

  \visible<4->{
  \begin{hlbox}
    \textbf{Fix} (Bertsekas 1982): cut the coupling with the \textbf{reduced Hessian} $H_R := \chi_{\mathcal{I}}\, H\, \chi_{\mathcal{I}} + \chi_{\mathcal{A}}$, i.e.\ Newton on the free set $\mathcal{I}$, a gradient step on the active set $\mathcal{A}$.
  \end{hlbox}
  }
\end{frame}

% ------------------------------------------------------------
\begin{frame}[label=pnalg]{The Projected Newton Method}
  \begin{hlbox}
    Given $z_0 \in Z_{\rm ad}$, for $k = 0, 1, 2, \dots$
    \begin{enumerate}
      \item<1-> Solve state and adjoint equations; form $g_k = \nabla\hat J(z_k)$. Stop if stationary.
      \item<2-> $\varepsilon$-active set, with $\varepsilon_k := \min\{\varepsilon, \norm{z_k - P_{Z_{\rm ad}}(z_k - g_k)}_{L^2}\}$:
        \[
          \mathcal{A}_k := \{ z_k \le z_a + \varepsilon_k,\, g_k > 0 \} \cup \{ z_k \ge z_b - \varepsilon_k,\, g_k < 0 \}, \quad \mathcal{I}_k := D \setminus \mathcal{A}_k .
        \]
      \item<3-> Solve $H_R\, d_k = g_k$ \emph{inexactly} by CG, with $\mathcal{A} = \mathcal{A}_k$, $\mathcal{I} = \mathcal{I}_k$ \hfill (next slide)
      \item<4-> Armijo along $z_k(s) := P_{Z_{\rm ad}}(z_k - s\,d_k)$: accept $s = \beta^m$ once
        \[
          \hat J(z_k) - \hat J(z_k(s)) \ge \sigma\Big( s\,(g_k, \chi_{\mathcal{I}_k} d_k)_{L^2} + \big(g_k, \chi_{\mathcal{A}_k}(z_k - z_k(s))\big)_{L^2} \Big).
        \]
      \item<5-> Set $z_{k+1} = z_k(s)$.
    \end{enumerate}
  \end{hlbox}
\end{frame}

% ------------------------------------------------------------
\begin{frame}{Matrix-Free Realization and Convergence}
  \begin{hlbox}
    On $\mathcal{A}_k$: $d_k = g_k$. On $\mathcal{I}_k$: \textbf{CG in the $L^2$ ($M$-) inner product}; each step costs one product $Hv$, i.e.\ two linear PDE solves (Section 5).
  \end{hlbox}

  \visible<2->{
  \begin{hlbox}
    \textbf{Spectrum again:} on $L^2(\mathcal{I}_k)$ the quadratic form of $\chi_{\mathcal{I}_k} H \chi_{\mathcal{I}_k}$ lies in $[\alpha, L]$, so CG counts are bounded in $h$. \textbf{Inexact:} stop CG at relative residual $\eta_k \to 0$ (Dembo--Eisenstat--Steihaug 1982).
  \end{hlbox}
  }

  \visible<3->{
  \begin{qbox}
    \textbf{Convergence:} global (Armijo along the arc), superlinear once the active set settles (Bertsekas 1982), mesh independent. \textbf{Relative:} primal--dual active sets $=$ semismooth Newton (Hinterm\"uller--Ito--Kunisch 2002).
  \end{qbox}
  }
\end{frame}
