\section{The State Equation}





\begin{frame}{Step 1: The State Equation}
  \begin{overlayarea}{\textwidth}{2.6cm}
  \only<1>{
  \begin{hlbox}
    \textbf{Strong form.} Given $z \in \Ltwo$, find $u$ such that
    \[
      \begin{aligned}
        -\Delta u &= z \;\;\text{in } D,\\
               u &= 0 \;\;\text{on } \partial D.
      \end{aligned}
    \]
  \end{hlbox}
  }
  \only<2->{
  \begin{hlbox}
    \textbf{Weak form.} Multiply by $v \in \Hzero$ and integrate by parts: find $u \in \Hzero$ with
    \[
      a(u,v) := \int_D \nabla u \cdot \nabla v \dd x = (z,v)_{L^2} \qquad \forall v \in \Hzero .
    \]
  \end{hlbox}
  }
  \end{overlayarea}

  \visible<3->{
  \begin{hlbox}
    \textbf{Operator form.} With $A : \Hzero \to H^{-1}(D)$, $\dual{Au}{v}_{H^{-1},H^1_0} := a(u,v)$, and the load map $\iota : \Ltwo \to H^{-1}(D)$, $\dual{\iota z}{v}_{H^{-1},H^1_0} := (z,v)_{L^2}$:
    \[
      Au = \iota z \quad \text{in } H^{-1}(D).
    \]
    {\small Subscripts drop the domain: $\norm{\cdot}_{L^2} := \norm{\cdot}_{L^2(D)}$, etc.}
  \end{hlbox}
  }
\end{frame}

\begin{frame}[label=Sprops]{Step 1: Well-Posedness and the Solution Operator}
  \begin{hlbox}
    \textbf{$a$ is bounded and coercive} (Friedrichs: $\norm{v}_{L^2} \le c_F\norm{\nabla v}_{L^2}$):
    \[
      |a(u,v)| \le \norm{u}_{H^1}\norm{v}_{H^1}, \qquad a(v,v) \ge \gamma\norm{v}_{H^1}^2, \quad \gamma := (1 + c_F^2)^{-1}.
    \]
  \end{hlbox}

  \visible<2->{
  \begin{hlbox}
    \textbf{Lax--Milgram:} for every $z \in \Ltwo$ there is a unique $u \in \Hzero$, and $\norm{u}_{H^1} \le \gamma^{-1}\norm{z}_{L^2}$. So
    \[
      S := A^{-1}\iota : \Ltwo \to \Hzero \quad \text{is linear and bounded, with } S'(z) = S .
    \]
  \end{hlbox}
  }

  \visible<3->{
  \begin{hlbox}
    \textbf{Compactness:} $\iota$ is compact (Rellich--Kondrachov and Schauder), so $S$ is compact.
  \end{hlbox}
  }

  \visible<4->{
  \begin{qbox}
    \textbf{Key consequence.} $z_n \weakto z$ in $\Ltwo$ $\Rightarrow$ $Sz_n \to Sz$ in $\Hzero$.
  \end{qbox}
  }
  \vfill\hfill\hyperlink{fadetails}{\beamergotobutton{Proofs}}
\end{frame}

\section{Existence of Optimal Controls}
% ============================================================
\begin{frame}{Step 2: The Reduced Problem}
  \begin{hlbox}
    Eliminate the state $u = Sz$. The tracking term observes $u$ in $\Ltwo$ through $C : \Hzero \hookrightarrow \Ltwo$:
    \[
      \min_{z \in Z_{\rm ad}}\; J(Sz,z) = \frac12 \big\| \underbrace{C}_{\text{observe}}\,\underbrace{S}_{\text{solve}} z - u_d \big\|_{L^2}^2 + \frac{\alpha}{2}\norm{z}_{L^2}^2 .
    \]
  \end{hlbox}

  \visible<2->{
  \begin{hlbox}
    \begin{itemize}
      \item $Z_{\rm ad}$ is convex and closed in $\Ltwo$; for $z_a \le z_b$ in $\Ltwo$ it is also nonempty and bounded.
      \item $z \mapsto J(Sz,z)$ is convex and continuous, since $CS$ is linear and bounded; $\alpha$-strongly convex if $\alpha > 0$.
    \end{itemize}
  \end{hlbox}
  }

  \visible<3->{
  \begin{qbox}
    We optimize over $z$ only, but every evaluation of $J(Sz,z)$ requires a PDE solve.
  \end{qbox}
  }
\end{frame}

\begin{frame}{Step 3: Existence via the Direct Method}
  \begin{qbox}
    \textbf{Theorem.} Let $\alpha \ge 0$ and let $Z_{\rm ad}$ be nonempty, convex, and closed. If $Z_{\rm ad}$ is bounded \emph{or} $\alpha > 0$, then the reduced problem has a solution $\bar z \in Z_{\rm ad}$. If $\alpha > 0$, it is unique.
  \end{qbox}

  \visible<2->{
  \begin{hlbox}
    \textbf{Why the alternative?} For $\alpha > 0$, $\norm{z}_{L^2}^2 \le \tfrac{2}{\alpha}J(Sz,z)$: sublevel sets are bounded. For $\alpha = 0$ and $Z_{\rm ad} = \Ltwo$, the infimum $0$ is attained only if $u_d \in \operatorname{ran}(CS) = \{ u \in \Hzero : \Delta u \in \Ltwo \}$.
  \end{hlbox}
  }

  \visible<3->{
  \begin{qbox}
    This is the ill-posedness of the inverse problem; the Tikhonov term $\tfrac{\alpha}{2}\norm{z}_{L^2}^2$ restores existence.
  \end{qbox}
  }
\end{frame}

\begin{frame}{Proof of Existence (1/2): A Weak Limit in $Z_{\rm ad}$}
  \begin{hlbox}
    \begin{enumerate}
      \item<1-> \textbf{Minimizing sequence.} $\bar\jmath := \inf_{Z_{\rm ad}} J(Sz,z) \ge 0$ is finite. Choose $z_n \in Z_{\rm ad}$ with $J(Sz_n,z_n) \to \bar\jmath$.
      \item<2-> \textbf{Boundedness.} If $Z_{\rm ad}$ is bounded, so is $(z_n)$. Otherwise $\alpha > 0$ and, for $n$ large,
        $\norm{z_n}_{L^2}^2 \le \tfrac{2}{\alpha}(\bar\jmath + 1)$.
      \item<3-> \textbf{Weak compactness.} $\Ltwo$ is a Hilbert space, so bounded sequences have weakly convergent subsequences: $z_n \weakto \bar z$ (after relabeling).
      \item<4-> \textbf{Feasibility.} $Z_{\rm ad}$ is convex and closed, hence weakly closed (Mazur). Thus $\bar z \in Z_{\rm ad}$.
    \end{enumerate}
  \end{hlbox}

  \visible<5->{
  \begin{qbox}
    It remains to show $J(S\bar z, \bar z) \le \liminf_n J(Sz_n, z_n) = \bar\jmath$: \textbf{weak lower semicontinuity}.
  \end{qbox}
  }
\end{frame}

\begin{frame}{Proof of Existence (2/2): Weak Lower Semicontinuity}
  \begin{hlbox}
    \textbf{Route A: convexity.} $z \mapsto J(Sz,z)$ is convex and continuous on $\Ltwo$. Its sublevel sets are convex and closed, hence weakly closed (Mazur): it is weakly l.s.c.
  \end{hlbox}

  \visible<2->{
  \begin{hlbox}
    \textbf{Route B: compactness of $S$.} By Step 1, $Sz_n \to S\bar z$ in $\Hzero$, so $CSz_n \to CS\bar z$ and the tracking term \emph{converges}. Only the control cost needs l.s.c., and norms are weakly l.s.c.: $\norm{\bar z}_{L^2} \le \liminf_n \norm{z_n}_{L^2}$.
  \end{hlbox}
  }

  \visible<3->{
  \begin{hlbox}
    Either way, $J(S\bar z,\bar z) \le \bar\jmath$, so $\bar z$ is optimal. \textbf{Uniqueness:} $g \circ CS$ is convex and $\tfrac{\alpha}{2}\norm{\cdot}_{L^2}^2$ is $\alpha$-strongly convex, so the sum has at most one minimizer. \hfill \qed
  \end{hlbox}
  }

  \visible<4->{
  \begin{qbox}
    Route A needs convexity, which fails for Example 3. Route B only needs compactness of $z \mapsto u$, which survives.
  \end{qbox}
  }
\end{frame}

\section{Optimality Conditions}
% ============================================================
\begin{frame}{Step 4: First-Order Necessary Conditions}
  \begin{qbox}
    \textbf{Theorem.} Let $Z_{\rm ad}$ be convex and $\hat J(z) := J(Sz,z)$ G\^ateaux differentiable at a local minimizer $\bar z$. Then
    \[
      \hat J'(\bar z)(z - \bar z) \ge 0 \qquad \forall z \in Z_{\rm ad}.
    \]
    For convex $\hat J$ it is sufficient: $\hat J(z) \ge \hat J(\bar z) + \hat J'(\bar z)(z - \bar z) \ge \hat J(\bar z)$.
  \end{qbox}

  \visible<2->{
  \begin{hlbox}
    \textbf{Proof.} $\bar z + t(z - \bar z) \in Z_{\rm ad}$ for $t \in (0,1]$. By optimality, for small $t$,
    \[
      0 \le \hat J\big(\bar z + t(z - \bar z)\big) - \hat J(\bar z) = t\,\hat J'(\bar z)(z - \bar z) + o(t) .
    \]
    \vspace{-1ex}Divide by $t$, let $t \downarrow 0$. \hfill \qed
  \end{hlbox}
  }

  \visible<3->{
  \begin{qbox}
    Only feasible directions enter: \textbf{G\^ateaux differentiability suffices}.
  \end{qbox}
  }
\end{frame}

% Step 5a: dual operators and Riesz maps

\begin{frame}{Step 5: Dual Operators and Riesz Maps}
  \framesubtitle{Following Lions (1971)}
  \begin{hlbox}
    \begin{itemize}
      \item<1-> \textbf{Dual operator} of $T \in \mathcal{L}(X,Y)$: \; $T^* \in \mathcal{L}(Y^*, X^*)$, \; $\dual{T^* y}{x}_{X^*,X} := \dual{y}{Tx}_{Y^*,Y}$.
      \item<2-> \textbf{Riesz map} of a Hilbert space $X$: \; $\Lambda_X : X \to X^*$, \; $\dual{\Lambda_X x}{y}_{X^*,X} := (x, y)_X$.
    \end{itemize}
    \visible<2->{We never identify $X$ with $X^*$: every Riesz map is written out.}
  \end{hlbox}

  \visible<3->{
  \begin{hlbox}
    \textbf{Our operators.} With $\Lambda := \Lambda_{L^2}$, the load map is $\iota = C^*\Lambda$, and
    \[
      CS = C \circ A^{-1} \circ \iota : \;\; \Ltwo \xrightarrow{\;\iota\;} H^{-1}(D) \xrightarrow{\;A^{-1}\;} \Hzero \xrightarrow{\;C\;} \Ltwo .
    \]
  \end{hlbox}
  }

  \visible<4->{
  \begin{qbox}
    \textbf{Discretely:} $A \to K$, \; $\iota, \Lambda \to M$, \; $C \to I$, \; $T^* \to T^\top$. Dropping $\Lambda$ or $\iota$ means replacing $M$ by $I$: the source of mesh dependence (Section 6).
  \end{qbox}
  }
\end{frame}

% ------------------------------------------------------------
% Step 5b: derivative, adjoint equation, gradient
% ------------------------------------------------------------
\begin{frame}{Step 5: The Adjoint Equation and the Gradient}
  \begin{hlbox}
    \textbf{Derivative.} \; $\hat J'(z) = (CS)^*\Lambda(CSz - u_d) + \alpha\Lambda z \in \Ltwo^*$. \; How do we evaluate $(CS)^*$?
  \end{hlbox}

  \visible<2->{
  \begin{hlbox}
    \textbf{Lemma.} $(CS)^*\Lambda(CSz - u_d) = \Lambda Cp$, where $p$ solves the \textbf{adjoint equation}
    \[
      -\Delta p = u - u_d \;\;\text{in } D, \qquad p = 0 \;\;\text{on } \partial D .
    \]
    \emph{Proof.} $(CS)^* = \iota^* A^{-*} C^*$, $A^* = A$, and $\dual{\iota^* p}{h}_{(L^2)^*,L^2} = (h, Cp)_{L^2}$. \hfill\qed
  \end{hlbox}
  }

  \visible<3->{
  \begin{qbox}
    \textbf{Gradient:} $\nabla\hat J(z) = \Lambda^{-1}\hat J'(z) = Cp + \alpha z$, one extra \emph{linear} PDE solve. Writing $p$ for $Cp$, the VI is \; $(\bar p + \alpha\bar z,\, z - \bar z)_{L^2} \ge 0$ for all $z \in Z_{\rm ad}$.
  \end{qbox}
  }
\end{frame}

\begin{frame}[label=kkt]{The Primal--Dual Optimality System}
  \begin{hlbox}
    \textbf{Theorem.} $\bar z \in Z_{\rm ad}$ is optimal if and only if there exist $\bar u, \bar p \in \Hzero$ such that
    \[
      \begin{aligned}
        -\Delta \bar u &= \bar z \;\;\text{in } D, \quad \bar u = 0 \;\;\text{on } \partial D && \text{(state)}\\
        -\Delta \bar p &= \bar u - u_d \;\;\text{in } D, \quad \bar p = 0 \;\;\text{on } \partial D && \text{(adjoint)}\\
        (\bar p + \alpha \bar z,\; z - \bar z)_{L^2} &\ge 0 \quad \forall z \in Z_{\rm ad} && \text{(VI)}
      \end{aligned}
    \]
  \end{hlbox}

  \visible<2->{
  \begin{hlbox}
    \textbf{Compare with $\R^n$.} For $\min J(u,z)$ s.t.\ $e(u,z) = 0$, $z \in K$, with $\Lag = J - \lambda^\top e$, the KKT conditions are
    \[
      e(\bar u,\bar z) = 0, \qquad \nabla_u \Lag = 0, \qquad \nabla_z \Lag^\top (z - \bar z) \ge 0 \;\; \forall z \in K .
    \]
  \end{hlbox}
  }

  \visible<3->{
  \begin{qbox}
    An infinite-dimensional KKT system: $\bar p$ is the \textbf{Lagrange multiplier} for the PDE.
  \end{qbox}
  }
  \vfill\hfill\hyperlink{lagrange}{\beamergotobutton{Formal Lagrange technique}}
\end{frame}

\begin{frame}[label=projform]{The Projection Formula}
  \begin{columns}[c]
    \begin{column}{0.58\textwidth}
      \begin{hlbox}
        $Z_{\rm ad}$ is defined pointwise, so the VI holds iff, a.e.\ with $g := \bar p + \alpha\bar z$,
        \[
          g \;\begin{cases}
            = 0 & \text{where } z_a < \bar z < z_b,\\
            \ge 0 & \text{where } z_a = \bar z < z_b,\\
            \le 0 & \text{where } z_a < \bar z = z_b.
          \end{cases}
        \]
      \end{hlbox}
    \end{column}
    \begin{column}{0.38\textwidth}
      \visible<2->{
      \begin{center}
      \begin{tikzpicture}[scale=0.8]
        \draw[-{Stealth[length=2mm]}, muted] (-0.3,0) -- (3.8,0) node[right, font=\footnotesize] {$w$};
        \draw[-{Stealth[length=2mm]}, muted] (0,-0.3) -- (0,3.0) node[above, font=\footnotesize] {$\bar z$};
        \draw[dashed, muted!60] (0,0.8) -- (3.6,0.8);
        \draw[dashed, muted!60] (0,2.2) -- (3.6,2.2);
        \node[left, font=\footnotesize] at (0,0.8) {$z_a$};
        \node[left, font=\footnotesize] at (0,2.2) {$z_b$};
        \draw[very thick, accent] (0.1,0.8) -- (0.8,0.8) -- (2.2,2.2) -- (3.5,2.2);
      \end{tikzpicture}
      \end{center}
      }
    \end{column}
  \end{columns}

  \visible<2->{
  \begin{qbox}
    Dividing by $\alpha > 0$: \; $\bar z = P_{[z_a,z_b]}\big(-\tfrac{1}{\alpha}\bar p\big) = \max\big\{z_a, \min\{z_b, -\tfrac{1}{\alpha}\bar p\}\big\}$ a.e., \; i.e.\ \; $\bar z = P_{Z_{\rm ad}}\big(\bar z - \tfrac{1}{\alpha}\nabla\hat J(\bar z)\big)$.
  \end{qbox}
  }
  \vfill\hfill\hyperlink{pwproof}{\beamergotobutton{Proof of the pointwise form}}
\end{frame}
